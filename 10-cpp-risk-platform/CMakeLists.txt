cmake_minimum_required(VERSION 3.16)
project(QuantRiskPlatform LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(quant INTERFACE)
target_include_directories(quant INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(credit_risk apps/credit.cpp)
add_executable(market_risk apps/market.cpp)
add_executable(liquidity_risk apps/liquidity.cpp)
foreach(app credit_risk market_risk liquidity_risk)
 target_link_libraries(${app} PRIVATE quant)
endforeach()
enable_testing()
add_executable(quant_tests tests/tests.cpp)
target_link_libraries(quant_tests PRIVATE quant)
add_test(NAME risk_tests COMMAND quant_tests)
