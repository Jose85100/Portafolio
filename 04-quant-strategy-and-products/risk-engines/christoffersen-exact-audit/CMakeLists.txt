cmake_minimum_required(VERSION 3.16)
project(ChristoffersenExactAudit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(target exact_audit exact_tests)
  if(target STREQUAL "exact_audit")
    add_executable(${target} src/main.cpp)
  else()
    add_executable(${target} tests/tests.cpp)
  endif()
  target_include_directories(${target} PRIVATE include)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
endforeach()
enable_testing()
add_test(NAME exact_contract COMMAND exact_tests)
