cmake_minimum_required(VERSION 3.16)
project(WeightedTailAudit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(AUDIT_SANITIZERS "Address and undefined-behavior sanitizers" OFF)
option(AUDIT_COVERAGE "GCC line coverage" OFF)
enable_testing()
add_executable(weighted_tail_cli src/main.cpp)
add_executable(weighted_tail_tests tests/test_weighted.cpp)
foreach(target weighted_tail_cli weighted_tail_tests)
  target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../include)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(AUDIT_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
  if(AUDIT_COVERAGE)
    target_compile_options(${target} PRIVATE --coverage -O0 -g)
    target_link_options(${target} PRIVATE --coverage)
  endif()
endforeach()
add_test(NAME weighted_invariants COMMAND weighted_tail_tests)
