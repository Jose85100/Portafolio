cmake_minimum_required(VERSION 3.16)
project(Stage3CashflowAudit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AUDIT_SANITIZERS "Address and undefined behavior checks" OFF)
foreach(target stage3_cli stage3_tests)
  if(target STREQUAL "stage3_cli")
    add_executable(${target} src/main.cpp)
  else()
    add_executable(${target} tests/tests.cpp)
  endif()
  target_include_directories(${target} PRIVATE ../include)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
  if(AUDIT_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
enable_testing()
add_test(NAME stage3_contract COMMAND stage3_tests)
