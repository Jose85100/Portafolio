cmake_minimum_required(VERSION 3.16)
project(christoffersen_sequential_audit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
include_directories(include ../christoffersen-exact-audit/include)
add_executable(sequential_audit src/main.cpp)
enable_testing()
add_executable(sequential_tests tests/tests.cpp)
add_test(NAME sequential_tests COMMAND sequential_tests)
