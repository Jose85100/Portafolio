cmake_minimum_required(VERSION 3.20)
project(moment_summary_transport LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UBSan" OFF)
add_executable(moments_consumer src/cpp/consumer.cpp)
add_executable(moments_cpp_tests src/cpp/test_core.cpp)
foreach(target moments_consumer moments_cpp_tests)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
  if(ENABLE_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
enable_testing()
add_test(NAME moments_core COMMAND moments_cpp_tests)
